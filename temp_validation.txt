"        function validateRequestSubmission(event) {"

"            const endDate = document.getElementById('requestEndDate').value;"  
"            "  
"            if (!startDate || !endDate) {"  
"                alert('Please fill in both start and end dates.');"

"                return false;"  
"            }"  
"            "  
"            const start = new Date(startDate);"  
"            const end = new Date(endDate);"  
"            const duration = Math.ceil((end - start) / (1000 * 60 * 60 * 24)) + 1;"  
"            "  
"            // Calculate current usage (only approved trips and previous trips)"  
"            const userData = JSON.parse(localStorage.getItem(`stiwUserData_${userEmail}`)) || {};"

"            const requests = userData.requests || [];"  
"            "

"            const twelveMonthsAgo = new Date();"  
"            twelveMonthsAgo.setFullYear(twelveMonthsAgo.getFullYear() - 1);"  
"            "  
"            // Count approved trips"  
"            allTrips.forEach(trip => {"  
"                const tripStart = new Date(trip.startDate);"  
"                if (tripStart >= twelveMonthsAgo) {"  
"                    currentUsage += trip.duration;"  
"                }"  
"            });"  
"            "  
"            // Count approved requests"  
"            requests.forEach(request => {"  
"                if ((request.status === 'Approved' || request.status === 'Approved (Auto-approved for shortening)')) {"  
"                    const requestStart = new Date(request.startDate);"  
"                    if (requestStart >= twelveMonthsAgo) {"  
"                        currentUsage += request.duration;"  
"                    }"  
"                }"  
"            });"  
"            "  
"            const projectedTotal = currentUsage + duration;"  
"            "  
"            if (projectedTotal > 45) {"

"            "

"            "
